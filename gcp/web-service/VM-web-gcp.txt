Configurar VM
# ----------------------------------------------------------

sudo apt-get update
sudo apt-get install -y docker.io docker-compose

gcloud auth configure-docker us-central1-docker.pkg.dev
TOKEN=$(gcloud auth print-access-token)
echo $TOKEN | sudo docker login -u oauth2accesstoken --password-stdin https://us-central1-docker.pkg.dev

Docker Compose
# ----------------------------------------------------------

nano docker-compose-web-gcp.yml
sudo docker-compose -f docker-compose-web-gcp.yml up -d
sudo docker ps

# Cloud Run BackEnd
# ---------------------

gcloud run deploy backend-service \
  --image=us-central1-docker.pkg.dev/desarrollo-cloud-457900/desarrollo-cloud/backend:latest \
  --platform=managed \
  --region=us-central1 \
  --allow-unauthenticated \
  --set-env-vars="DATABASE_URL=postgresql+psycopg2://postgres:admin@34.68.52.115:5432/USER,CLOUD_STORAGE=bucket-documents-users"
